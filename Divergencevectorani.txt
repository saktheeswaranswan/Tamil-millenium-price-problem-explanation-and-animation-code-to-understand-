import java.util.ArrayList;

// APDE 3D Vector Divergence Simulator - Advanced Tiny Area Flux Edition
int divState = 1; // 1 = Positive (Source), -1 = Negative (Sink), 0 = Zero Div
boolean showVectors = true; 

// 0 = Center Mode, 1 = Tiny Area Mode, 2 = Grid Mode
int viewMode = 1; 

float sizeSlider = 0.8; // Controls the scale of the solids
float subdivSlider = 3.0; // Controls the "Step Faces" (tiny area squares)
float rx = 0.5;
float ry = -0.5;

ArrayList<CustomSolid> solids = new ArrayList<CustomSolid>();
int currentSolidIndex = 0;

void setup() {
  size(displayWidth, displayHeight, P3D);
  orientation(LANDSCAPE);
  
  // 1. Standard Geometry
  solids.add(new CustomSolid("Cube", buildCube()));
  solids.add(new CustomSolid("Sphere", buildSphere(10)));
  solids.add(new CustomSolid("Torus", buildTorus(14, 6, 1.0, 0.45)));
  solids.add(new CustomSolid("Cylinder", buildPrism(16, 1.0, 1.2)));
  solids.add(new CustomSolid("Octagonal Prism", buildPrism(8, 1.2, 1.0)));
  solids.add(new CustomSolid("Tetrahedron", buildTetrahedron()));
  solids.add(new CustomSolid("Octahedron", buildOctahedron()));
  solids.add(new CustomSolid("Wedge", buildWedge()));

  // 2. Biological / Procedural Geometry
  solids.add(new CustomSolid("Pumpkin", buildBioShape(0, 14)));
  solids.add(new CustomSolid("Tomato", buildBioShape(1, 14)));
  solids.add(new CustomSolid("Potato", buildBioShape(2, 14)));
  solids.add(new CustomSolid("Brinjal / Eggplant", buildBioShape(3, 14)));

  // 3. Composite Mesh Geometry
  solids.add(new CustomSolid("3D Human", buildHuman()));
  solids.add(new CustomSolid("Car", buildCar()));
  solids.add(new CustomSolid("Truck", buildTruck()));
}

void draw() {
  background(15);

  // --- 1. Draw 3D Scene ---
  pushMatrix();
  translate(width / 2, height / 2, 0);

  // Handle touch rotation
  if (mousePressed && mouseY > 120 && mouseY < height - 250) {
    rx += (mouseY - pmouseY) * 0.01;
    ry -= (mouseX - pmouseX) * 0.01;
  }
  rotateX(rx);
  rotateY(ry);

  CustomSolid currentSolid = solids.get(currentSolidIndex);

  if (viewMode == 0 || viewMode == 1) {
    drawSingleMode(currentSolid);
  } else {
    drawGridMode(currentSolid);
  }
  
  popMatrix();

  // --- 2. Draw 2D UI Overlay ---
  hint(DISABLE_DEPTH_TEST);
  camera();
  noLights();
  drawUI(currentSolid.name);
  hint(ENABLE_DEPTH_TEST);
}

// --- RENDERING MODES ---

void drawSingleMode(CustomSolid solid) {
  float currentSize = height * 0.25 * sizeSlider; 
  int subdivs = (viewMode == 1) ? (int)subdivSlider : 1;
  
  solid.drawSolid(currentSize, divState, viewMode, subdivs);
  
  if (showVectors) {
    solid.drawVectors(currentSize, divState, viewMode, subdivs);
  }
}

void drawGridMode(CustomSolid solid) {
  int resolution = 3; 
  float fieldSize = height * 0.6;
  float step = fieldSize / (resolution - 1);
  float offset = -fieldSize / 2;
  float currentSize = step * 0.3 * sizeSlider; 

  for (int i = 0; i < resolution; i++) {
    for (int j = 0; j < resolution; j++) {
      for (int k = 0; k < resolution; k++) {
        float x = offset + i * step;
        float y = offset + j * step;
        float z = offset + k * step;

        pushMatrix();
        translate(x, y, z);
        solid.drawSolid(currentSize, divState, 2, 1);
        if (showVectors) {
          solid.drawVectors(currentSize, divState, 2, 1); 
        }
        popMatrix();
      }
    }
  }
}

// --- 3D GEOMETRY ENGINE ---

class Face {
  PVector[] vertices;
  PVector center;
  PVector normal;

  Face(PVector[] v) {
    this(v, new PVector(0, 0, 0)); 
  }

  Face(PVector[] v, PVector refCenter) {
    this.vertices = v;
    center = new PVector();
    for (PVector p : v) center.add(p);
    center.div(v.length);

    PVector v0 = PVector.sub(v[1], v[0]);
    PVector v1 = PVector.sub(v[2], v[0]);
    normal = v0.cross(v1).normalize();

    if (normal.dot(PVector.sub(center, refCenter)) < 0) {
      normal.mult(-1); 
    }
  }
  
  // Actually subdivides the face into smaller discrete Face objects (tiny squares/triangles)
  ArrayList<Face> getSubFaces(int steps) {
    ArrayList<Face> subFaces = new ArrayList<Face>();
    if (steps <= 1) {
      subFaces.add(this);
      return subFaces;
    }
    
    if (vertices.length == 4) {
      for (int i = 0; i < steps; i++) {
        for (int j = 0; j < steps; j++) {
          float u1 = (float)i / steps;       float u2 = (float)(i + 1) / steps;
          float v1 = (float)j / steps;       float v2 = (float)(j + 1) / steps;
          
          PVector p1 = bilerp(vertices[0], vertices[1], vertices[2], vertices[3], u1, v1);
          PVector p2 = bilerp(vertices[0], vertices[1], vertices[2], vertices[3], u2, v1);
          PVector p3 = bilerp(vertices[0], vertices[1], vertices[2], vertices[3], u2, v2);
          PVector p4 = bilerp(vertices[0], vertices[1], vertices[2], vertices[3], u1, v2);
          
          Face sf = new Face(new PVector[]{p1, p2, p3, p4});
          sf.normal = this.normal.copy();
          subFaces.add(sf);
        }
      }
    } else if (vertices.length == 3) {
      for (int i = 0; i < steps; i++) {
        for (int j = 0; j < steps - i; j++) {
          PVector p1 = triLerp(vertices[0], vertices[1], vertices[2], i, j, steps);
          PVector p2 = triLerp(vertices[0], vertices[1], vertices[2], i+1, j, steps);
          PVector p3 = triLerp(vertices[0], vertices[1], vertices[2], i, j+1, steps);
          Face sf1 = new Face(new PVector[]{p1, p2, p3});
          sf1.normal = this.normal.copy();
          subFaces.add(sf1);
          
          if (i + 1 + j < steps) {
            PVector p4 = triLerp(vertices[0], vertices[1], vertices[2], i+1, j+1, steps);
            Face sf2 = new Face(new PVector[]{p2, p4, p3});
            sf2.normal = this.normal.copy();
            subFaces.add(sf2);
          }
        }
      }
    } else {
      subFaces.add(this);
    }
    return subFaces;
  }
  
  PVector bilerp(PVector v0, PVector v1, PVector v2, PVector v3, float u, float v) {
    PVector pBottom = PVector.lerp(v0, v1, u);
    PVector pTop = PVector.lerp(v3, v2, u);
    return PVector.lerp(pBottom, pTop, v);
  }
  
  PVector triLerp(PVector v0, PVector v1, PVector v2, int i, int j, int steps) {
    float u = (float)i / steps;
    float v = (float)j / steps;
    float w = 1.0f - u - v;
    PVector p = new PVector();
    p.add(PVector.mult(v0, u));
    p.add(PVector.mult(v1, v));
    p.add(PVector.mult(v2, w));
    return p;
  }
}

class CustomSolid {
  String name;
  ArrayList<Face> faces;

  CustomSolid(String name, ArrayList<Face> faces) {
    this.name = name;
    this.faces = faces;
  }

  void drawSolid(float scaleFactor, int state, int mode, int subdiv) {
    // "Blink" animation using sine wave based on time
    float blinkAlpha = (mode == 1) ? (sin(millis() * 0.006f) * 40 + 130) : 45;
    PVector zeroField = new PVector(1, 0.5, -0.2).normalize();
    
    for (Face f : faces) {
      ArrayList<Face> renderFaces = new ArrayList<Face>();
      if (mode == 1) {
        renderFaces = f.getSubFaces(subdiv);
      } else {
        renderFaces.add(f);
      }
      
      for (Face rf : renderFaces) {
        pushMatrix();
        
        if (mode == 1) {
          // Color coding for Tiny Area mode based on flux
          if (state == 1) fill(255, 60, 60, blinkAlpha); // Source (Red)
          else if (state == -1) fill(60, 200, 255, blinkAlpha); // Sink (Cyan)
          else {
            float dot = rf.normal.dot(zeroField);
            if (dot > 0) fill(255, 230, 50, blinkAlpha); // Flux Out = Yellow
            else fill(50, 255, 80, blinkAlpha); // Flux In = Green
          }
          
          stroke(255, 100);
          strokeWeight(1.0);
          
          // "Step orthogonally" - Shrink slightly towards center to show grid tiles clearly
          PVector c = PVector.mult(rf.center, scaleFactor);
          translate(c.x, c.y, c.z);
          scale(0.85f); // 15% gap between squares
          translate(-c.x, -c.y, -c.z);
          
        } else {
          fill(100, 150, 255, 45); 
          stroke(150, 200, 255, 120);
          strokeWeight(1.5);
        }
        
        beginShape();
        for (PVector v : rf.vertices) {
          vertex(v.x * scaleFactor, v.y * scaleFactor, v.z * scaleFactor);
        }
        endShape(CLOSE);
        
        popMatrix();
      }
    }
  }

  void drawVectors(float scaleFactor, int state, int mode, int subdiv) {
    PVector zeroField = new PVector(1, 0.5, -0.2).normalize(); 
    float baseT = (millis() % 1500) / 1500.0f; 
    
    float vLen = scaleFactor * 0.8f; 
    if (mode == 1) vLen = max(scaleFactor * 0.3f, scaleFactor * 1.2f / subdiv); // Scale vector length for tiny squares

    for (Face f : faces) {
      ArrayList<Face> activeFaces = new ArrayList<Face>();
      if (mode == 1) activeFaces = f.getSubFaces(subdiv);
      else activeFaces.add(f);
      
      for (Face rf : activeFaces) {
        PVector c = PVector.mult(rf.center, scaleFactor);
        PVector n = rf.normal;
        
        float dot = n.dot(zeroField);
        boolean isExiting = dot > 0;
        
        int numArrows = (mode == 0) ? 3 : 1; 
        
        for (int i = 0; i < numArrows; i++) {
          float t = (baseT + (float)i / numArrows) % 1.0f;
          float arrowScale = (mode == 1) ? max(0.4f, 1.5f / subdiv) : 1.2f;
          
          if (mode == 0) { 
            // --- CENTER MODE ---
            PVector center = new PVector(0,0,0);
            PVector dir = c.copy().normalize();
            float dist = c.mag();
            
            if (state == 1) { 
              PVector pos = PVector.mult(dir, t * (dist + vLen));
              drawAnimatedArrow(pos, dir, state, isExiting, arrowScale);
            } else if (state == -1) { 
              PVector pos = PVector.mult(dir, (1.0f - t) * (dist + vLen));
              drawAnimatedArrow(pos, PVector.mult(dir, -1), state, isExiting, arrowScale);
            } else { 
              PVector offset = PVector.sub(c, PVector.mult(zeroField, c.dot(zeroField)));
              PVector startPos = PVector.add(offset, PVector.mult(zeroField, -scaleFactor * 1.5));
              PVector endPos = PVector.add(offset, PVector.mult(zeroField, scaleFactor * 1.5));
              PVector pos = PVector.lerp(startPos, endPos, t);
              drawAnimatedArrow(pos, zeroField, state, isExiting, arrowScale);
            }
            
          } else { 
            // --- TINY AREA / GRID MODE ---
            if (state == 1) { 
              // Source: Animate popping OUT
              PVector pos = PVector.add(c, PVector.mult(n, t * vLen));
              drawAnimatedArrow(pos, n, state, isExiting, arrowScale);
            } else if (state == -1) { 
              // Sink: Animate going IN
              PVector pos = PVector.add(c, PVector.mult(n, (1.0f - t) * vLen));
              drawAnimatedArrow(pos, PVector.mult(n, -1), state, isExiting, arrowScale);
            } else { 
              // Zero Flux: Flow straight through the square (Doesn't contribute)
              PVector startPos = PVector.sub(c, PVector.mult(zeroField, vLen * 0.5f));
              PVector endPos = PVector.add(c, PVector.mult(zeroField, vLen * 0.5f));
              PVector pos = PVector.lerp(startPos, endPos, t);
              drawAnimatedArrow(pos, zeroField, state, isExiting, arrowScale);
            }
          }
        }
      }
    }
  }
}

// --- SHAPE BUILDERS ---

ArrayList<Face> buildBioShape(int type, int res) {
  ArrayList<Face> faces = new ArrayList<Face>();
  for (int i = 0; i < res; i++) {
    float lon1 = map(i, 0, res, -PI, PI), lon2 = map(i + 1, 0, res, -PI, PI);
    for (int j = 0; j < res; j++) {
      float lat1 = map(j, 0, res, -HALF_PI + 0.01, HALF_PI - 0.01);
      float lat2 = map(j + 1, 0, res, -HALF_PI + 0.01, HALF_PI - 0.01);
      faces.add(new Face(new PVector[]{
        evalBio(type, lat1, lon1), evalBio(type, lat1, lon2),
        evalBio(type, lat2, lon2), evalBio(type, lat2, lon1)
      }));
    }
  }
  return faces;
}

PVector evalBio(int type, float lat, float lon) {
  float r = 1.0, x = 0, y = 0, z = 0;
  if (type == 0) { r = 1.0 + 0.12 * cos(10 * lon); x = r * cos(lat) * cos(lon); y = r * sin(lat) * 0.65; z = r * cos(lat) * sin(lon); }
  else if (type == 1) { r = 1.0 + 0.04 * cos(6 * lon); x = r * cos(lat) * cos(lon); y = r * sin(lat) * 0.85; z = r * cos(lat) * sin(lon); }
  else if (type == 2) { r = 1.0 + 0.08 * sin(3 * lon) * cos(4 * lat) + 0.05 * cos(5 * lon); x = r * cos(lat) * cos(lon) * 1.3; y = r * sin(lat) * 0.9; z = r * cos(lat) * sin(lon) * 1.0; }
  else if (type == 3) { r = 1.0 + 0.4 * sin(lat); x = r * cos(lat) * cos(lon) * 0.6 + sin(lat) * 0.2; y = sin(lat) * 1.4; z = r * cos(lat) * sin(lon) * 0.6; }
  return new PVector(x, y, z);
}

ArrayList<Face> buildTranslatedBox(float w, float h, float d, float tx, float ty, float tz) {
  ArrayList<Face> faces = new ArrayList<Face>();
  float hw = w/2, hh = h/2, hd = d/2;
  PVector ref = new PVector(tx, ty, tz); 
  PVector[] v = {
    new PVector(-hw+tx, -hh+ty,  hd+tz), new PVector( hw+tx, -hh+ty,  hd+tz),
    new PVector( hw+tx,  hh+ty,  hd+tz), new PVector(-hw+tx,  hh+ty,  hd+tz),
    new PVector(-hw+tx, -hh+ty, -hd+tz), new PVector( hw+tx, -hh+ty, -hd+tz),
    new PVector( hw+tx,  hh+ty, -hd+tz), new PVector(-hw+tx,  hh+ty, -hd+tz)
  };
  faces.add(new Face(new PVector[]{v[0], v[1], v[2], v[3]}, ref)); faces.add(new Face(new PVector[]{v[5], v[4], v[7], v[6]}, ref)); 
  faces.add(new Face(new PVector[]{v[4], v[0], v[3], v[7]}, ref)); faces.add(new Face(new PVector[]{v[1], v[5], v[6], v[2]}, ref)); 
  faces.add(new Face(new PVector[]{v[4], v[5], v[1], v[0]}, ref)); faces.add(new Face(new PVector[]{v[3], v[2], v[6], v[7]}, ref)); 
  return faces;
}

}

ArrayList<Face> buildCube() { return buildTranslatedBox(2, 2, 2, 0, 0, 0); }
ArrayList<Face> buildSphere(float ignore) { return buildBioShape(5, 12); }

ArrayList<Face> buildPrism(int sides, float r, float h) {
  ArrayList<Face> faces = new ArrayList<Face>();
  PVector[] top = new PVector[sides], bot = new PVector[sides];
  for (int i = 0; i < sides; i++) {
    float a = TWO_PI * i / sides;
    top[i] = new PVector(r*cos(a), -h, r*sin(a)); bot[i] = new PVector(r*cos(a), h, r*sin(a));
  }
  faces.add(new Face(top)); faces.add(new Face(bot));
  for (int i = 0; i < sides; i++) {
    int nxt = (i + 1) % sides;
    faces.add(new Face(new PVector[]{ top[i], top[nxt], bot[nxt], bot[i] }));
  }
  return faces;
}

ArrayList<Face> buildTorus(int resU, int resV, float R, float r) {
  ArrayList<Face> faces = new ArrayList<Face>();
  for (int i = 0; i < resU; i++) {
    float u1 = map(i, 0, resU, 0, TWO_PI), u2 = map(i + 1, 0, resU, 0, TWO_PI);
    for (int j = 0; j < resV; j++) {
      float v1 = map(j, 0, resV, 0, TWO_PI), v2 = map(j + 1, 0, resV, 0, TWO_PI);
      PVector p1 = new PVector((R+r*cos(v1))*cos(u1), r*sin(v1), (R+r*cos(v1))*sin(u1));
      PVector p2 = new PVector((R+r*cos(v1))*cos(u2), r*sin(v1), (R+r*cos(v1))*sin(u2));
      PVector p3 = new PVector((R+r*cos(v2))*cos(u2), r*sin(v2), (R+r*cos(v2))*sin(u2));
      PVector p4 = new PVector((R+r*cos(v2))*cos(u1), r*sin(v2), (R+r*cos(v2))*sin(u1));
      PVector tubeCenter = new PVector(R*cos((u1+u2)/2), 0, R*sin((u1+u2)/2));
      faces.add(new Face(new PVector[]{p1, p2, p3, p4}, tubeCenter));
    }
  }
  return faces;
}

ArrayList<Face> buildTetrahedron() {
  ArrayList<Face> faces = new ArrayList<Face>();
  PVector v1 = new PVector(1.2, 1.2, 1.2), v2 = new PVector(1.2, -1.2, -1.2);
  PVector v3 = new PVector(-1.2, 1.2, -1.2), v4 = new PVector(-1.2, -1.2, 1.2);
  faces.add(new Face(new PVector[]{v1, v2, v3})); faces.add(new Face(new PVector[]{v1, v3, v4}));
  faces.add(new Face(new PVector[]{v1, v4, v2})); faces.add(new Face(new PVector[]{v2, v4, v3}));
  return faces;
}

ArrayList<Face> buildOctahedron() {
  ArrayList<Face> faces = new ArrayList<Face>();
  PVector top = new PVector(0,-1.5,0), bot = new PVector(0,1.5,0);
  PVector f = new PVector(0,0,1.5), b = new PVector(0,0,-1.5), l = new PVector(-1.5,0,0), r = new PVector(1.5,0,0);
  faces.add(new Face(new PVector[]{top, f, r})); faces.add(new Face(new PVector[]{top, r, b}));
  faces.add(new Face(new PVector[]{top, b, l})); faces.add(new Face(new PVector[]{top, l, f}));
  faces.add(new Face(new PVector[]{bot, r, f})); faces.add(new Face(new PVector[]{bot, b, r}));
  faces.add(new Face(new PVector[]{bot, l, b})); faces.add(new Face(new PVector[]{bot, f, l}));
  return faces;
}

ArrayList<Face> buildWedge() {
  ArrayList<Face> faces = new ArrayList<Face>();
  PVector[] v = {
    new PVector(-1,-1, 1), new PVector(1,-1, 1), new PVector(-1,-1,-1),
    new PVector(-1, 1, 1), new PVector(1, 1, 1), new PVector(-1, 1,-1)
  };
  faces.add(new Face(new PVector[]{v[0], v[1], v[2]})); faces.add(new Face(new PVector[]{v[3], v[5], v[4]})); 
  faces.add(new Face(new PVector[]{v[0], v[3], v[4], v[1]})); faces.add(new Face(new PVector[]{v[1], v[4], v[5], v[2]})); 
  faces.add(new Face(new PVector[]{v[2], v[5], v[3], v[0]})); 
  return faces;
}

// --- RENDERING HELPERS ---

void drawAnimatedArrow(PVector pos, PVector dir, int state, boolean isExiting, float scaleW) {
  if (dir.mag() < 0.001) return;
  PVector d = dir.copy().normalize();
  
  pushMatrix();
  translate(pos.x, pos.y, pos.z);

  PVector zAxis = new PVector(0, 0, 1);
  PVector crossAxis = zAxis.cross(d);
  float angle = PVector.angleBetween(zAxis, d);

  if (crossAxis.mag() > 0.001) {
    crossAxis.normalize();
    rotate(angle, crossAxis.x, crossAxis.y, crossAxis.z);
  } else if (d.z < 0) { 
    rotateX(PI);
  }

  float arrowLen = 12 * scaleW; 
  float tailLen = 15 * scaleW;
  float r = 5 * scaleW;         

  if (state == 1) { // Source (Red)
    fill(255, 60, 60); stroke(255, 0, 0, 150);
  } else if (state == -1) { // Sink (Cyan)
    fill(60, 200, 255); stroke(0, 150, 255, 150);
  } else { // Zero Flux
    if (isExiting) { // Exiting = Yellow
      fill(255, 230, 50); stroke(220, 200, 0, 150);
    } else { // Entering = Green
      fill(50, 255, 80); stroke(0, 200, 0, 150);
    }
  }

  strokeWeight(3 * scaleW);
  line(0, 0, -tailLen, 0, 0, 0);

  noStroke();
  drawArrowHead(r, arrowLen);
  
  popMatrix();
}

void drawArrowHead(float r, float arrowLen) {
  int sides = 8; 
  beginShape(TRIANGLES);
  for (int i = 0; i < sides; i++) {
    float a1 = TWO_PI * i / sides;
    float a2 = TWO_PI * (i + 1) / sides;
    vertex(0, 0, 0); 
    vertex(r * cos(a1), r * sin(a1), -arrowLen);
    vertex(r * cos(a2), r * sin(a2), -arrowLen);
    vertex(0, 0, -arrowLen);
    vertex(r * cos(a2), r * sin(a2), -arrowLen);
    vertex(r * cos(a1), r * sin(a1), -arrowLen);
  }
  endShape();
}

void drawUI(String solidName) {
  // Top Buttons
  int btnW = width / 3;
  int btnH = 120;
  textSize(32); textAlign(CENTER, CENTER);

  fill(divState == 1 ? (showVectors ? color(200, 30, 30) : color(100, 20, 20)) : color(60));
  rect(0, 0, btnW, btnH);
  fill(255); text("Source (+)" + (divState == 1 && !showVectors ? " (OFF)" : ""), btnW / 2, btnH / 2);

  fill(divState == -1 ? (showVectors ? color(30, 150, 200) : color(20, 80, 100)) : color(60));
  rect(btnW, 0, btnW, btnH);
  fill(255); text("Sink (-)" + (divState == -1 && !showVectors ? " (OFF)" : ""), btnW + btnW / 2, btnH / 2);

  fill(divState == 0 ? (showVectors ? color(200, 200, 40) : color(100, 100, 20)) : color(60));
  rect(btnW * 2, 0, btnW, btnH);
  fill(255); text("Flux Zero (0)" + (divState == 0 && !showVectors ? " (OFF)" : ""), btnW * 2 + btnW / 2, btnH / 2);

  // Bottom Control Panel
  // 1. Toggle View Mode Button
  fill(100, 150, 255); stroke(255); strokeWeight(3);
  rect(width / 2 - 200, height - 250, 400, 70, 15);
  fill(255); noStroke();
  String modeText = viewMode == 0 ? "Switch to TINY AREA MODE" : (viewMode == 1 ? "Switch to GRID MODE" : "Switch to CENTER MODE");
  text(modeText, width / 2, height - 215);

  // 2. Shape Cycle Buttons
  fill(80); stroke(150); strokeWeight(3);
  rect(20, height - 160, 220, 90, 15);
  rect(width - 240, height - 160, 220, 90, 15);
  fill(255); noStroke();
  text("< Prev Shape", 130, height - 115);
  text("Next Shape >", width - 130, height - 115);
  text("Shape: " + solidName, width / 2, height - 150);
  
  // 3. Dual Sliders
  float sliderMinX = width * 0.3;
  float sliderMaxX = width * 0.7;
  
  // Size Slider
  float sizeY = height - 90;
  stroke(150); strokeWeight(8);
  line(sliderMinX, sizeY, sliderMaxX, sizeY);
  float sliderX1 = map(sizeSlider, 0.2, 1.5, sliderMinX, sliderMaxX);
  fill(255); noStroke(); ellipse(sliderX1, sizeY, 40, 40);
  textSize(20); text("Shape Size", sliderMinX - 80, sizeY);
  
  // Step Faces / Subdiv Slider (Only active in Tiny Area Mode)
  float subY = height - 40;
  if (viewMode == 1) {
    stroke(150); strokeWeight(8);
    line(sliderMinX, subY, sliderMaxX, subY);
    float sliderX2 = map(subdivSlider, 1, 6, sliderMinX, sliderMaxX);
    fill(255, 200, 100); noStroke(); ellipse(sliderX2, subY, 40, 40);
    fill(255); textSize(20); text("Step Faces", sliderMinX - 80, subY);
  }
}

// --- INTERACTION LOGIC ---

void mousePressed() { handleUI(); }

void mouseDragged() { handleUI(); }

void mouseReleased() {
  if (mouseY <= 120) {
    int btnW = width / 3;
    int clickedState = (mouseX < btnW) ? 1 : ((mouseX < btnW * 2) ? -1 : 0);
    
    if (divState == clickedState) showVectors = !showVectors;
    else {
      divState = clickedState;
      showVectors = true;
    }
  }
  
  if (mouseY > height - 250 && mouseY < height - 180) {
    if (mouseX > width / 2 - 200 && mouseX < width / 2 + 200) {
      viewMode = (viewMode + 1) % 3;
    }
  }

  if (mouseY > height - 160 && mouseY < height - 70) {
    if (mouseX > 20 && mouseX < 240) currentSolidIndex = (currentSolidIndex - 1 + solids.size()) % solids.size();
    else if (mouseX > width - 240 && mouseX < width - 20) currentSolidIndex = (currentSolidIndex + 1) % solids.size();
  }
}

void handleUI() {
  float sliderMinX = width * 0.3;
  float sliderMaxX = width * 0.7;
  
  if (mouseY > height - 120 && mouseY < height - 70 && mouseX > sliderMinX - 20 && mouseX < sliderMaxX + 20) {
    sizeSlider = constrain(map(mouseX, sliderMinX, sliderMaxX, 0.2, 1.5), 0.2, 1.5);
  }
  
  if (viewMode == 1 && mouseY > height - 70 && mouseY < height - 10 && mouseX > sliderMinX - 20 && mouseX < sliderMaxX + 20) {
    subdivSlider = constrain(map(mouseX, sliderMinX, sliderMaxX, 1, 6), 1, 6);
  }
}
